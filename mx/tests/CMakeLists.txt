cmake_minimum_required(VERSION 3.25)
project(MxVPNImportTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
set(CMAKE_AUTOMOC ON)
enable_testing()
add_executable(mx_import_tests importTests.cpp)
target_link_libraries(mx_import_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_include_directories(mx_import_tests PRIVATE ../../client)
add_test(NAME mx_import_tests COMMAND mx_import_tests)
add_executable(mx_traffic_tests trafficTests.cpp)
target_link_libraries(mx_traffic_tests PRIVATE Qt6::Core Qt6::Test)
target_include_directories(mx_traffic_tests PRIVATE ../../client)
add_test(NAME mx_traffic_tests COMMAND mx_traffic_tests)
add_executable(mx_lookup_tests lookupTests.cpp)
target_link_libraries(mx_lookup_tests PRIVATE Qt6::Core Qt6::Network Qt6::Test)
target_include_directories(mx_lookup_tests PRIVATE ../../client)
add_test(NAME mx_lookup_tests COMMAND mx_lookup_tests)
add_executable(mx_validate_config validateConfig.cpp)
target_link_libraries(mx_validate_config PRIVATE Qt6::Core Qt6::Network)
target_include_directories(mx_validate_config PRIVATE ../../client)
if(APPLE)
